  export async function buildUserProfile(user, socialProofs): Promise<MappedUserProfile> {
    const tokenBalance = await getTokenBalanceSafe(user);
    // Role lookup must never break sign-in: a DB failure degrades to no roles,
    // which only hides UI. Authorization is unaffected — it re-reads grants
    // server-side on every privileged request.
    const grantsResult = await listActiveGrants(user.id);
    const roles = grantsResult.isOk()

    return transformToProfile(user, socialProofs, tokenBalance, roles);
  }
