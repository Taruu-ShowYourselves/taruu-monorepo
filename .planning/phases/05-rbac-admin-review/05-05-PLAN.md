Behavior, in order:
1. Validate the space: `const validSpaces = [...MUNICIPALITIES, KNESSET_SCOPE]`. If
   `!validSpaces.includes(input.spaceId)` return `errAsync(validation(['spaceId: unknown space']))`.
2. `findLatestApplicationForUser(userId)`. If the returned row is non-null AND its
   `status === 'submitted'`, return
   `errAsync(conflict('An application is already under review'))`. If its
   `status === 'approved'` and its `space_id === input.spaceId`, return
   `errAsync(conflict('You already hold an approved application for this space'))`.
   Any other state (null, `rejected`, `withdrawn`, approved-for-another-space) proceeds.
3. `insertApplication({ user_id: userId, space_id: input.spaceId, motivation: input.motivation,
   contact_phone: input.contactPhone ?? null, evidence_urls: input.evidenceUrls, status: 'submitted' })`.
   Map a Postgres unique-violation on `uq_cm_applications_open` (error code `23505`, visible in the
   `DB` AppError's `cause`) to `conflict('An application is already under review')` so the race
   between step 2 and step 3 does not surface as a 500.
4. `insertAuditEvent({ subject_type: 'community_manager_application', subject_id: application.id,
   event: 'submitted', subject_user_id: userId, actor_user_id: userId, role: 'community_manager',
   space_id: input.spaceId, reason: null, detail: null })`.
5. `.map(() => ({ application: toManagerApplication(application) }))`.

**b. Route.** Create `apps/web/src/app/api/manager-applications/route.ts`:

```ts
/**
 * /api/manager-applications — applicant-facing (RBAC-03).
 *
 * POST  submit a community-manager application for one space
 * GET   the caller's own latest application + their own live grants
 *
 * This route is NOT privileged: any signed-in user may apply. The privileged
 * side lives under /api/admin/manager-applications and goes through
 * requireReviewAuthority().
 */
import { NextRequest, NextResponse } from 'next/server';
import { errAsync } from 'neverthrow';
import { SubmitManagerApplicationRequestSchema } from '@sync/shared/contracts';
import { getSessionFromRequest } from '@/services/auth/session';
import { managerApplicationLimiter, createRateLimitResponse } from '@/lib/rate-limit';
import { parse, respond } from '@/server/http/respond';
import { unauthorized } from '@/server/http/errors';
import {
  getOwnApplicationState,
  submitManagerApplication,
} from '@/server/app/authz/submit-application';

export async function POST(request: NextRequest) {
  const session = await getSessionFromRequest(request);
  if (!session) return respond(errAsync(unauthorized()));

  const limit = await managerApplicationLimiter.check(session.userId);
  if (limit.limited) return createRateLimitResponse(limit) as NextResponse;

  const body = await request.json().catch(() => null);
  const parsed = parse(SubmitManagerApplicationRequestSchema, body);
  if (parsed.isErr()) return respond(errAsync(parsed.error));

  return respond(submitManagerApplication({}, session.userId, parsed.value), { status: 201 });
}

export async function GET(request: NextRequest) {
  const session = await getSessionFromRequest(request);
  if (!session) return respond(errAsync(unauthorized()));
  return respond(getOwnApplicationState(session.userId));
}
```
