```ts
/**
 * RLS test harness (RLS-04) — the repo's first automated proof that a Row Level
 * Security policy actually filters rows.
 *
 * Why this exists: before Phase 5, `public.user_id()` was never fed, so every
 * per-user policy in this database matched zero rows and nobody could tell,
 * because nothing exercised them. Plan 05-01 built the transport. This module
 * is how we keep it honest.
 *
 * Design for reuse: Phase 7 (MIG-01..04) migrates 25 more tables onto this
 * transport and needs the same proof for each. Everything table-agnostic lives
 * here; a per-table test file should be a few dozen lines of assertions.
 *
 * Safety: every row this harness creates is tagged with RLS_TEST_MARKER and
 * deleted in cleanup. It never reads, updates or deletes a row it did not
 * create, and it refuses to run under NODE_ENV=production.
 */

import { describe } from 'vitest';
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import type { Database } from '@/lib/supabase/types';
import { mintSupabaseAccessToken } from '@/lib/supabase/user-token';

/** Every seeded row carries this so cleanup can be exact. */
export const RLS_TEST_MARKER = 'rls-harness';

export interface RlsTestEnv {
  url: string;
  anonKey: string;
  serviceRoleKey: string;
  jwtSecret: string;
}

/**
 * Returns the four credentials the harness needs, or null when any is missing.
 * Missing credentials mean "skip", never "fail" — CI has no database.
 */
export function readRlsTestEnv(): RlsTestEnv | null {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const anonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;
  const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY;
  const jwtSecret = process.env.SUPABASE_JWT_SECRET;
  if (!url || !anonKey || !serviceRoleKey || !jwtSecret) return null;
  if (process.env.NODE_ENV === 'production') return null;
  return { url, anonKey, serviceRoleKey, jwtSecret };
}

const SKIP_NOTE =
  'RLS suite skipped: set NEXT_PUBLIC_SUPABASE_URL, NEXT_PUBLIC_SUPABASE_ANON_KEY, ' +
  'SUPABASE_SERVICE_ROLE_KEY and SUPABASE_JWT_SECRET, then run\n' +
  '  cd apps/web && set -a && . ./.dev.vars && set +a && npx vitest run src/__tests__/rls\n' +
  'See apps/web/docs/RLS-TESTING.md.';

/**
 * `describe` that skips (loudly) when there is no database to talk to.
 * Use this instead of bare `describe` in every *.rls.test.ts file.
 */
export const describeRls: typeof describe.skip = (() => {
  const env = readRlsTestEnv();
  if (!env) {
    // eslint-disable-next-line no-console
    console.warn(SKIP_NOTE);
    return describe.skip;
  }
  return describe;
})();

const NO_SESSION = { auth: { persistSession: false, autoRefreshToken: false } } as const;

/** Anon key, no token. This is what an attacker with the public key gets. */
export function anonClient(env: RlsTestEnv): SupabaseClient<Database> {
  return createClient<Database>(env.url, env.anonKey, NO_SESSION);
}

/** Service role. Bypasses RLS — used ONLY to seed and clean up, never to assert. */
export function serviceClient(env: RlsTestEnv): SupabaseClient<Database> {
  return createClient<Database>(env.url, env.serviceRoleKey, NO_SESSION);
}

/**
 * Anon key + a freshly minted token for `userId`. This is the client under test
 * and is built the same way `createUserScopedClient()` builds it, with the
 * credentials passed explicitly so a test can point at a non-default project.
 */
export function rlsUserClient(env: RlsTestEnv, userId: string): SupabaseClient<Database> {
  return createClient<Database>(env.url, env.anonKey, {
    ...NO_SESSION,
    accessToken: () => mintSupabaseAccessToken(userId),
  });
}

export interface ThrowawayUser {
  id: string;
  email: string;
}

export interface SeededFixture {
  userA: ThrowawayUser;
  userB: ThrowawayUser;
  /** Two real municipalities.code values, read from the database. */
  spaceA: string;
  spaceB: string;
  /** Deletes every row this fixture created. Always call in afterAll. */
  cleanup: () => Promise<void>;
}

/**
 * Creates two throwaway users and resolves two real space codes.
 *
 * Emails are `rls-harness+<uuid>@taruu.invalid` — `.invalid` is reserved by
 * RFC 2606 and can never be a real address, so a leaked fixture cannot mail a
 * human.
 */
export async function seedThrowawayUsers(env: RlsTestEnv): Promise<SeededFixture> {
  const svc = serviceClient(env);

  const { data: spaces, error: spaceError } = await svc
    .from('municipalities')
    .select('code')
    .order('code')
    .limit(2);
  if (spaceError) throw spaceError;
  if (!spaces || spaces.length < 2) {
    throw new Error('RLS harness needs at least two rows in municipalities');
  }

  const mkEmail = () => `${RLS_TEST_MARKER}+${crypto.randomUUID()}@taruu.invalid`;
  const { data: users, error: userError } = await svc
    .from('users')
    .insert([{ email: mkEmail() }, { email: mkEmail() }])
    .select('id, email');
  if (userError) throw userError;
  if (!users || users.length !== 2) throw new Error('RLS harness failed to seed two users');

  const [userA, userB] = users as ThrowawayUser[];

  return {
    userA,
    userB,
    spaceA: spaces[0].code,
    spaceB: spaces[1].code,
    cleanup: async () => {
      const ids = [userA.id, userB.id];
      // role_grant_events has no FK and an append-only trigger, so its rows are
      // deliberately NOT deleted — that is what append-only means. They are
      // marked with RLS_TEST_MARKER in `reason` and are harmless.
      await svc.from('community_manager_applications').delete().in('user_id', ids);
      await svc.from('role_grants').delete().in('user_id', ids);
      await svc.from('users').delete().in('id', ids);
    },
  };
}

/**
 * The anon key must read nothing from `table`. PostgREST answers an RLS-denied
 * SELECT with 200 + [], not an error, so assert on emptiness AND on the absence
 * of rows rather than on a thrown error.
 */
export async function expectAnonReadsNothing(
  env: RlsTestEnv,
  table: 'role_grants' | 'community_manager_applications' | 'role_grant_events'
): Promise<void> {
  const { data, error } = await anonClient(env).from(table).select('id');
  if (error) return; // an outright permission error is also an acceptable denial
  if (data && data.length > 0) {
    throw new Error(`anon key read ${data.length} row(s) from ${table} — RLS is not denying it`);
  }
}

/**
 * `userId` must see `expectedVisibleIds` and must NOT see `expectedHiddenIds`.
 * A 42P17 from Postgres ("infinite recursion detected in policy") surfaces here
 * as a query error and is re-thrown with that context, because it is the single
 * most likely way these policies break.
 */
export async function expectCrossUserInvisible(
  env: RlsTestEnv,
  table: 'role_grants' | 'community_manager_applications' | 'role_grant_events',
  userId: string,
  expectedVisibleIds: string[],
  expectedHiddenIds: string[]
): Promise<void> {
  const { data, error } = await rlsUserClient(env, userId).from(table).select('id');
  if (error) {
    throw new Error(
      `${table} read as ${userId} failed (${error.code ?? '?'}): ${error.message}` +
        (error.code === '42P17' ? ' — a policy is recursing; route the role lookup through a SECURITY DEFINER helper' : '')
    );
  }
  const seen = new Set((data ?? []).map((r) => r.id));
  for (const id of expectedVisibleIds) {
    if (!seen.has(id)) throw new Error(`${table}: ${userId} should see ${id} but does not`);
  }
  for (const id of expectedHiddenIds) {
    if (seen.has(id)) throw new Error(`${table}: ${userId} must NOT see ${id} but does`);
  }
}
```
