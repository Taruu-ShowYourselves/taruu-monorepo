```ts
/**
 * Campaign push fan-out. Runs OFF the request path: the in-app rows are the
 * source of truth and were already written, so a push failure degrades the
 * delivery rather than losing the notification.
 */
export async function fanOutCampaignPush(
  campaign: { id: string; space_id: string; title: string; body: string },
  userIds: string[]
): Promise<void> {
  try {
    const { activeTokensForUsers } = await import('@/server/infra/supabase/push.repo');
    const tokens = await activeTokensForUsers(userIds).match((t) => t, () => []);
    if (tokens.length === 0) return;
    await sendBatchNotifications(tokens, {
      title: campaign.title,
      body: campaign.body,
      data: { type: 'space_announcement', spaceId: campaign.space_id, campaignId: campaign.id },
      channelId: 'votes',
      priority: 'default',
    });
    // Record push-channel deliveries; failure here is logged, never thrown.
  } catch (error) {
    logger.warn('space campaign push fan-out failed (non-fatal)', { campaignId: campaign.id, error: String(error) });
  }
}
```
