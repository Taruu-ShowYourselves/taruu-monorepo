}

export function requireRole(
  userId: string, role: RoleName, spaceId: string | null
): ResultAsync<GrantFacts, AppError> {
  return findLiveGrant(userId, role, spaceId).andThen((row) =>
    billingRequirementSatisfied(userId, role, spaceId).andThen((billingActive) => {
      const decision = evaluateAuthorization({ grant: row ? toGrantFacts(row) : null, billingActive });
      return decision.allowed ? okAsync(decision.grant) : errAsync(forbidden(decision.reason));
    })
  );
}

// requireAdminScope reads listActiveGrants

```ts
/**
 * The billing prerequisite (MGR-01). Phase 5 left this a stable `true`; this is
 * the one line Phase 6 changes in the authorization layer, exactly as that
 * phase's header comment promised. requireRole's signature and every call site
 * are untouched.
 *
 * ONLY `community_manager` is billed. `super_admin` and `space_admin` are
 * platform roles with no subscription — gating them would 403 the Phase 5 review
 * console for every human including the bootstrapped super admin, which is the
 * single most damaging mistake available here. The early return is not an
 * optimisation; it is the guard.
 *
 * Authorization is still never DERIVED from payment state: a settled charge
 * cannot create a grant (manager_subscriptions.role_grant_id is NOT NULL and
 * points at an already-approved grant). Billing can only ever subtract.
 */
function billingRequirementSatisfied(
  userId: string,
  role: RoleName,
  spaceId: string | null
): ResultAsync<boolean, AppError> {
  if (role !== 'community_manager') return okAsync(true);
  // A community_manager grant always carries a space (role_grants_scope_ck).
  // A null here means a malformed grant; deny rather than guess.
  if (spaceId === null) return okAsync(false);

  return findLiveSubscription(userId, spaceId).map((row) =>
    isBillingActive(
      row
        ? { status: row.status, graceUntil: row.grace_until ? new Date(row.grace_until) : null }
        : null,
      new Date()
    )
  );
}
```

Two exports:

```ts
export function toSubscriptionSummary(
  row: ManagerSubscription,
  paymentMethod: BillingPaymentMethod | null
): ManagerSubscriptionSummary;

export function getManagerBillingStatus(
  userId: string,
  spaceId: string
): ResultAsync<ManagerBillingStatusResponse, AppError>;
```

`toSubscriptionSummary` maps snake_case row to the contract, and for `paymentMethod` emits only
`{ brand, last4 }` — never the token id, never the expiry, never the row itself. A card token in a
JSON response is a token in a browser's network log.

`getManagerBillingStatus`:
1. `findLiveSubscription(userId, spaceId)`
2. on `null` ⇒ `okAsync({ subscription: null, accessActive: false, charges: [] })`. No subscription
   is not access, and the endpoint says so plainly rather than 404ing — the panel needs to render
   the "you have no subscription" state.
3. otherwise resolve the payment method (`findPaymentMethodById` when `payment_method_id` is set)
   and `listChargesForSubscription(sub.id, 12)` — twelve months of history, capped
4. `accessActive` = `isBillingActive({ status, graceUntil }, new Date())`
5. map charges to `ManagerChargeSummary`, exposing `provider_document_id` as `documentId` (the
   receipt the user is entitled to see) but **not** `provider_charge_id`, `idempotency_key`, or
   `failure_code` — internal reconciliation detail, not user-facing

This use-case does **not** call `requireRole`. It is the caller's own data, scoped by `userId` from
the verified session, and a manager whose billing lapsed must still be able to look at the page that
explains why. Gating the status endpoint on active billing would lock people out of the screen that
tells them how to fix their billing. Say that in a comment — it looks like a missing authorization
check and it is not.
