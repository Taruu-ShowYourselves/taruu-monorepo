```ts
/**
 * Charge idempotency keys (MGR-03, and the SEC-04 discipline applied to billing).
 *
 * The key is a pure function of (subscription, period, attempt). It is generated
 * SERVER-side and is deterministic: never Date.now(), never a random value, never
 * anything a client supplied. That is what makes claim-before-charge work — two
 * concurrent renewal runs build the same key, the database's UNIQUE constraint
 * picks one winner, and the loser reads the winner's row instead of charging.
 */
export const CHARGE_KEY_PREFIX = 'mgrsub';

export function buildChargeIdempotencyKey(
  subscriptionId: string,
  periodStart: Date,
  attempt: number
): string {
  const day = periodStart.toISOString().slice(0, 10); // YYYY-MM-DD, UTC
  return `${CHARGE_KEY_PREFIX}:${subscriptionId}:${day}:${attempt}`;
}
```

- **Agorot in, ILS out.** Every internal amount is agorot; Green Invoice's `sum` is ILS. Convert
  once, at the boundary, in this file: `sum: input.amountAgorot / 100`. Do not let agorot leak into
  a `sum` field or ILS leak into a database column.
- **`idempotencyKey` rides in `custom`** on the token charge, mirroring how the merch and vote flows

  batch and leave every later subscription unprocessed.
