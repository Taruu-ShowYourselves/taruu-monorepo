```ts
/**
 * Minimum-cohort privacy for authority aggregates (AUTH-03).
 *
 * An aggregate computed over fewer than MIN_COHORT_SIZE residents is WITHHELD:
 * the returned value carries no number at all. Not rounded, not bucketed, not
 * zero — because a rounded figure over a cohort of three still identifies three
 * people, and a zero is indistinguishable from a real zero.
 *
 * SCOPE. This floor governs aggregates the authority dashboard INTRODUCES:
 * satisfaction broken down by any cut, engagement segments, response-rate cuts,
 * per-vote slices, and every CSV export. It does NOT retroactively suppress data
 * that is already public by product design — vote tallies, participant counts and
 * the municipality profile's satisfaction average are on the public council page
 * today and stay exactly as they are. Applying a floor to an already-public number
 * would be incoherent: the same figure visible to an anonymous visitor and hidden
 * from the authority.
 */

/** The floor, chosen 2026-08-03 (08-CONTEXT decision 5). Inclusive: n = 10 is released. */
export const MIN_COHORT_SIZE = 10;

export type Aggregate<T> =
  | { withheld: true; cohortSize: number; minCohortSize: number }
  | { withheld: false; cohortSize: number; value: T };

export function applyCohortFloor<T>(
  value: T,
  cohortSize: number,
  minCohortSize: number = MIN_COHORT_SIZE
): Aggregate<T> {
  if (cohortSize < minCohortSize) {
    // Note what is NOT here: no `value` key at all. A withheld aggregate must not
    // be serializable back into an approximation of the suppressed number.
    return { withheld: true, cohortSize, minCohortSize };
  }
  return { withheld: false, cohortSize, value };
}

export function isWithheld<T>(
  aggregate: Aggregate<T>
): aggregate is { withheld: true; cohortSize: number; minCohortSize: number } {
  return aggregate.withheld;
}
```
